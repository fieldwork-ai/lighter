"""Two additional deletion observations per version, retaining the initial preparation."""
from pathlib import Path
import os,subprocess as sp,json,hashlib,signal,shutil,csv
root=Path.cwd();out=Path('$HOME/git/lighter/.logs/051/hybrid-m5/deletion-followup-a1');out.mkdir(exist_ok=False)
new=root/'target/release/examples/lighter-bench';old=Path('$HOME/git/lighter/.logs/051/hybrid-m5/baseline-050/lighter-bench')
oldguest=Path('$HOME/.lighter/releases/0.5.0/share/lighter');guest=root/'guest/out'
source=sp.check_output(['git','rev-parse','HEAD'],text=True).strip();runtime=sp.check_output(['git','rev-parse','b78dfeb'],text=True).strip();baseline=sp.check_output(['git','rev-parse','v0.5.0^{commit}'],text=True).strip()
def digest(p):
 with p.open('rb') as f:return hashlib.file_digest(f,'sha256').hexdigest()
artifacts={str(p):digest(p) for p in [Path(__file__),Path(__file__).with_name('run-rm.sh'),root/'benchmarks/run.sh',new,old,guest/'Image',guest/'rootfs.ext4',oldguest/'Image',oldguest/'rootfs.ext4']}
(out/'environment.json').write_text(json.dumps(dict(source=source,runtime_source=runtime,baseline_source=baseline,artifacts=artifacts,order=['hybrid','050'],profile=dict(cpus=8,memory_mib=16384,disk_gib=128),repetitions=2,warmup='All three package-manager warm-ups; one untimed ripgrep, find-walk and copy precede two timed deletions. Initial deletion difference above 5% triggers these two additional observations only.',cases='rm-rf'),indent=2)+'\n')
env={k:v for k,v in os.environ.items() if not k.startswith('LIGHTER_') and k not in ('DOCKER_HOST','DOCKER_CONTEXT','DOCKER_CONFIG')}
env.update(PATH='$HOME/.orbstack/bin:/opt/homebrew/bin:$HOME/.cargo/bin:'+env['PATH'],BENCH_CPUS='8',BENCH_MEMORY_MIB='16384',BENCH_DISK_GIB='128',BENCH_EXTRA_WARM_CASES='npm-install pnpm-install yarn-install',BENCH_TOOLS_PATH='$HOME/git/lighter/.logs/050/tools/native/bin',BENCH_REQUIRE_PINNED_TOOLS='1',LIGHTER_BENCH_IMAGE_DIR='$HOME/git/lighter/.logs/050/benchmark-images',LIGHTER_BENCH_KEEP_OUTPUT='1')
report=(root/'benchmarks/RESULTS.md').read_bytes()
def interrupt(sig,frame):raise KeyboardInterrupt(f'signal {sig}')
signal.signal(signal.SIGTERM,interrupt)
success=False
(out/'phase').write_text('baseline\n')
monitor=sp.Popen(['python3','scripts/records/monitor-host.py',str(out)],env=env)
try:
 for version in ['hybrid','050']:
  sp.run(['git','diff','--quiet','HEAD'],check=True)
  assert all(digest(Path(p))==sha for p,sha in artifacts.items())
  label='051-m5-deletion-followup-a1-'+version;(out/'phase').write_text(label)
  runenv=dict(env,LIGHTER_BENCH_BIN=str(new if version=='hybrid' else old),LIGHTER_BENCH_GUEST_DIR=str(guest if version=='hybrid' else oldguest),LIGHTER_BENCH_SOURCE_SHA=runtime if version=='hybrid' else baseline,LIGHTER_DEMAND_RAM='1' if version=='hybrid' else '0',LIGHTER_DEMAND_BASE='1' if version=='hybrid' else '0',LIGHTER_BACKGROUND_RAM='1' if version=='hybrid' else '0')
  with (out/(label+'.log')).open('x') as log:
   p=sp.Popen(['python3','benchmarks/guard.py','--quiet','--target','lighter','--timeout','1800','--log',str(out/(label+'-guard.jsonl')),'--','bash',str(Path(__file__).with_name('run-rm.sh')),'--target','lighter','--label',label,'--reps','2','--cases','rm-rf'],env=runenv,stdout=log,stderr=sp.STDOUT)
   try:code=p.wait()
   finally:
    if p.poll() is None:p.terminate();p.wait(timeout=30)
  (root/'benchmarks/RESULTS.md').write_bytes(report)
  for suffix in ['.csv','.tree']:
   path=root/'benchmarks/results'/(label+suffix)
   if path.exists():shutil.copy2(path,out/path.name)
  shutil.copy2(root/'benchmarks/results/lighter-boot.log',out/(label+'-boot.log'))
  for diagnostic in (root/'.logs').glob('*'+label+'*'):
   if diagnostic.is_file() and diagnostic.suffix in ['.out','.log']:
    shutil.copy2(diagnostic,out/(diagnostic.name+'.log' if diagnostic.suffix=='.out' else diagnostic.name))
  if monitor.poll() is not None:raise RuntimeError('observer exited')
  if code:raise RuntimeError(f'{label} exit {code}')
  rows=list(csv.DictReader((out/(label+'.csv')).open()))
  assert len(rows)==2 and all(row['case']=='rm-rf' for row in rows)
  assert [row['rep'] for row in rows]==['1','2']
  for preparation in ['ripgrep','find-walk','copy-tree']:
   text=(root/'.logs'/f'deletion-preparation-{label}-{preparation}.out').read_text()
   assert sum(line.startswith('TIME_MS ') for line in text.splitlines())==1,preparation
  print('PASS',version,flush=True)
 assert all(digest(Path(p))==sha for p,sha in artifacts.items())
 success=True
finally:
 monitor.terminate()
 try:monitor.wait(timeout=10)
 except sp.TimeoutExpired:monitor.kill();monitor.wait()
 (root/'benchmarks/RESULTS.md').write_bytes(report)
 (out/'exit-code').write_text('0\n' if success else '1\n')
