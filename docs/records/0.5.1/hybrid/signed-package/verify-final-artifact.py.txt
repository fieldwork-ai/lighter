def digest(path):
 with path.open('rb') as f:return hashlib.file_digest(f,'sha256').hexdigest()
def linkedit(path):
 data=path.read_bytes()
 assert struct.unpack_from('<II',data)==(0xfeedfacf,0x100000c), 'expected arm64 Mach-O'
 offset=32; found=[]
 for _ in range(struct.unpack_from('<I',data,16)[0]):
  command,size=struct.unpack_from('<II',data,offset)
  assert size>=8 and offset+size<=len(data)
  if command==0x19 and data[offset+8:offset+24].rstrip(b'\0')==b'__LINKEDIT':
   vmaddr,vmsize,fileoff,filesize=struct.unpack_from('<QQQQ',data,offset+24)
   assert struct.unpack_from('<iiI',data,offset+56)==(1,1,0)
   found.append(dict(offset=offset,vmsize=vmsize,filesize=filesize,fileoff=fileoff))
  offset+=size
 assert len(found)==1
 return found[0]
def strip_and_normalize(path):
 original=linkedit(path)
 rounded=lambda size:(size+16383)//16384*16384
 assert original['vmsize']==rounded(original['filesize']), 'unexpected signed LINKEDIT mapping'
 sp.run(['codesign','--remove-signature',str(path)],check=True)
 unsigned=linkedit(path)
 assert unsigned['offset']==original['offset'] and unsigned['fileoff']==original['fileoff']
 assert unsigned['vmsize']==original['vmsize'], 'unexpected signature-removal mapping change'
 data=bytearray(path.read_bytes())
 # codesign retains the signed mapping size. Normalize only this derived,
 # read-only LINKEDIT field; every other remaining byte must match.
 struct.pack_into('<Q',data,unsigned['offset']+32,rounded(unsigned['filesize']))
 path.write_bytes(data)
 return dict(signed_vmsize=original['vmsize'],signed_filesize=original['filesize'],unsigned_filesize=unsigned['filesize'],canonical_vmsize=rounded(unsigned['filesize']))
