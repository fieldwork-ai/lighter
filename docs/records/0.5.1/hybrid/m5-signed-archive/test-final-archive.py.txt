"""Serial exact-archive checks without changing the daily installation."""
import argparse,hashlib,json,os,time
from pathlib import Path
import subprocess as sp

p=argparse.ArgumentParser();p.add_argument('--host',choices=['m1','m5'],required=True);p.add_argument('--out',type=Path,required=True);p.add_argument('--archive',type=Path,required=True);p.add_argument('--bootstrap',type=Path,required=True);p.add_argument('--legacy-dir',type=Path,required=True);p.add_argument('--helpers',type=Path,required=True);a=p.parse_args()
root=Path.cwd();base=a.helpers.resolve();out=a.out.resolve();out.mkdir(parents=True,exist_ok=False)
archive=a.archive.resolve();helper=a.bootstrap.resolve()
env={k:v for k,v in os.environ.items() if not k.startswith('LIGHTER_') and k not in ('DOCKER_HOST','DOCKER_CONTEXT','DOCKER_CONFIG')}
env['PATH']=str(Path.home()/'.orbstack/bin')+':/opt/homebrew/bin:'+str(Path.home()/'.cargo/bin')+':'+env['PATH']
def digest(path):
 with path.open('rb') as f:return hashlib.file_digest(f,'sha256').hexdigest()
inputs=[archive,helper,a.legacy_dir.resolve()/'lighter-0.4.1-arm64.tar.gz',a.legacy_dir.resolve()/'lighter-0.5.0-arm64.tar.gz',base/'fixtures/future-good.tar.gz',base/'fixtures/future-bad.tar.gz']
hashes={str(p):digest(p) for p in inputs}
daily=Path.home()/'.lighter';daily_state={}
for name in ['config.json','machine.identity']:
 if (daily/name).exists():daily_state[name]=digest(daily/name)
(out/'environment.json').write_text(json.dumps(dict(host=a.host,artifacts=hashes,daily_before=daily_state,source=sp.check_output(['git','rev-parse','HEAD'],text=True).strip()),indent=2)+'\n')
results=[];success=False
def call(label,command):
 (out/'phase').write_text(label+'\n');start=time.monotonic()
 with (out/(label+'.log')).open('x') as log:
  result=sp.run([str(x) for x in command],env=env,stdout=log,stderr=sp.STDOUT,timeout=1800)
 results.append(dict(stage=label,exit_code=result.returncode,seconds=round(time.monotonic()-start,2)))
 (out/'results.json').write_text(json.dumps(results,indent=2)+'\n');result.check_returncode();print('PASS',label,flush=True)
try:
 call('fresh-smoke',['bash',base/'smoke-signed.sh',archive,out/'smoke'])
 call('migration-050',['python3',base/'test-signed-upgrade.py','--old',inputs[3],'--new',archive,'--bootstrap',helper,'--out',out/'migration-050'])
 call('migration-041',['bash','scripts/test-release-upgrade.sh',inputs[2],archive,helper])
 call('activation-rollback',['bash','scripts/test-staged-upgrade.sh',archive,inputs[4],inputs[5],helper])
 call('development-update-lifecycle',['bash','scripts/test-update-lifecycle.sh'])
 call('signed-kind',['python3',base/'test-signed-kind.py','--archive',archive,'--out',out/'kind','--kind',root/('.logs/kind-'+a.host+'/bin/kind')])
 for path,sha in hashes.items():assert digest(Path(path))==sha,path
 for name,sha in daily_state.items():assert digest(daily/name)==sha,'daily state changed: '+name
 success=True
finally:
 (out/'exit-code').write_text('0\n' if success else '1\n')
